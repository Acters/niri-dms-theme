cmake_minimum_required(VERSION 3.22)
project(DrawyGlassStyle VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(drawyglassstyle MODULE drawyglassstyle.cpp)
set_target_properties(drawyglassstyle PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
target_link_libraries(drawyglassstyle PRIVATE Qt6::Widgets)
add_executable(style-test style-test.cpp)
target_link_libraries(style-test PRIVATE Qt6::Widgets)
