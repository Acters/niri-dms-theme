cmake_minimum_required(VERSION 3.22)
project(NiriGlassStyle VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(niriglassstyle MODULE niriglassstyle.cpp)
set_target_properties(niriglassstyle PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
target_link_libraries(niriglassstyle PRIVATE Qt6::Widgets)
add_executable(style-test style-test.cpp)
target_link_libraries(style-test PRIVATE Qt6::Widgets)
add_dependencies(style-test niriglassstyle)
add_executable(okular-viewport-test okular-viewport-test.cpp)
target_link_libraries(okular-viewport-test PRIVATE Qt6::Widgets)
add_dependencies(okular-viewport-test niriglassstyle)
enable_testing()
add_test(NAME niri-glass-native COMMAND style-test)
add_test(NAME niri-glass-readable COMMAND style-test --readability)
add_test(NAME niri-glass-okular-viewport COMMAND okular-viewport-test)
set_tests_properties(niri-glass-native niri-glass-readable niri-glass-okular-viewport PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins")
